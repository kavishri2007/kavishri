EXAMPLE(AMSTRONG NUMBER)

import java.util.*;
public class Palindrome {
    public static void main(String args[]) {
        Scanner sc = new Scanner(System.in);
        int n=sc.nextInt();
        int ans=n;
        int digit=0;
        int res=0;
        while(n>0) {
            n=n/10;
            digit++;
        }
        n=ans;
        while(n>0) {
            int r=n%10;
            int mul=1;
            for(int i=1; i<=digit; i++) {
                mul=mul*r;
            }
            res = res+mul;
            n=n/10;
        }
        System.out.println((res==ans)?"amstrong":"Not");
    }
}

        if(n<999) {
            System.out.print("Invalid");

             int last = n%100;
             int first = n;
             while(first>99) {
                 first = first /10;
             }
             System.out.print(last + first);
        }
